# 1. Проверить что miniapp файлы есть на сервере
ls -la /opt/impulse-bot/miniapp/

# 2. Установить зависимости
cd /opt/impulse-bot && npm install express 2>/dev/null

# 3. Обновить nginx (скопировать весь блок cat ... NGINX из DEPLOY_MINIAPP_PASTE.txt)

# 4. Запустить miniapp через PM2
cd /opt/impulse-bot && pm2 start pm2.config.js --only impulse-miniapp

# 5. Проверить статус
pm2 status

# 6. Проверить health check
curl -s http://localhost:3001/health

# 7. Проверить через nginx
curl -s http://109.69.22.112/miniapp/health
